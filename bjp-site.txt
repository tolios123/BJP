  <script>
    // Drawer Logic
    const drawer = document.getElementById('categoryDrawer');
    const overlay = document.getElementById('drawerOverlay');
    const openBtn = document.getElementById('openDrawer');
    const heroOpenBtn = document.getElementById('heroOpenDrawer');
    const closeBtn = document.getElementById('closeDrawer');
    const drawerCloseLinks = document.querySelectorAll('.drawer-close');

    function openDrawer() {
      drawer.classList.remove('-translate-x-full');
      overlay.classList.remove('hidden');
      setTimeout(() => overlay.classList.add('opacity-100'), 10);
    }

    function closeDrawer() {
      drawer.classList.add('-translate-x-full');
      overlay.classList.remove('opacity-100');
      setTimeout(() => overlay.classList.add('hidden'), 300);
    }

    openBtn.addEventListener('click', openDrawer);
    if(heroOpenBtn) heroOpenBtn.addEventListener('click', openDrawer);
    closeBtn.addEventListener('click', closeDrawer);
    overlay.addEventListener('click', closeDrawer);

    drawerCloseLinks.forEach(link => {
      link.addEventListener('click', closeDrawer);
    });

    // About Us Modal Logic
    const aboutModal = document.getElementById('aboutModal');
    const openAboutModalBtn = document.getElementById('openAboutModal');
    const closeAboutModalBtn = document.getElementById('closeAboutModal');
    const footerAboutBtn = document.getElementById('footerAboutBtn');

    function openAboutModal() {
      closeDrawer();
      aboutModal.classList.remove('hidden');
    }

    function closeAboutModal() {
      aboutModal.classList.add('hidden');
    }

    openAboutModalBtn.addEventListener('click', openAboutModal);
    if(footerAboutBtn) footerAboutBtn.addEventListener('click', openAboutModal);
    closeAboutModalBtn.addEventListener('click', closeAboutModal);

    aboutModal.addEventListener('click', (e) => {
      if (e.target === aboutModal) closeAboutModal();
    });

    // Category Buttons open Drawer
    document.querySelectorAll('.category-btn').forEach(btn => {
      btn.addEventListener('click', openDrawer);
    });
  </script>
